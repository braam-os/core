# Binary size budgets, in bytes. One "name = limit" per line; the name is the
# binary's file name. Raising a number is a deliberate act — see Release_Notes.md.
kernel.wasm = 262144

# The boot archive: every binary, plus what ships beside them. It is awaited
# before the first prompt, so it is boot latency as well as bytes — and it is
# where §4.4's duplication shows up, one runtime copy per program.
bundle.bin = 1048576
